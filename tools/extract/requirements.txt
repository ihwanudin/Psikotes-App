# Dependencies for tools/extract/ (F0 instrument-data extraction + its tests).
# Scoped to this internal tooling only - never imported by application code.
openpyxl==3.1.5

# pypdf chosen over PyMuPDF: BSD-licensed (PyMuPDF is AGPL); handles owner-password
# PDF decryption and per-glyph text-object coordinates, which is what the column-aware
# extractors (RMIB, IST WA) need. Revisit only if a subtest genuinely needs something
# pypdf cannot do (documented inline where that happens).
pypdf==6.19.0

# pdfminer.six (MIT licensed): the PAPI item PDF has no second structured source like
# RMIB's xlsx, so the coordinator required two genuinely independent extraction
# methods diffed word-for-word for the 90 statement pairs. pdftotext (poppler, C++,
# external CLI) is one; pdfminer.six is a separate pure-Python implementation used as
# the other - pypdf's layout mode fails outright on this PDF ("Rotated text
# discovered", empty output), and its default mode has the same mid-word phantom-space
# defect pypdf's coordinate extraction had for RMIB before switching to layout mode.
pdfminer.six==20260107

# IST FA/WU (image subtests): Pillow for compositing the page's raster bands
# and cropping; numpy for the dark-pixel mask; scipy for connected-component
# analysis, used two ways - detecting each item's/legend option's natural
# figure boundary (a uniform grid division was tried first and produced
# content bounding boxes landing exactly on the arbitrary grid line, a sign
# of clipping rather than a real edge - see the PR's report), and the
# required check that no crop's edge cuts through a dark region. All three
# are BSD-licensed, no PyMuPDF anywhere in this path.
Pillow==12.3.0
numpy==2.4.0
scipy==1.18.1
